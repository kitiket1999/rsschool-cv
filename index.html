function mute_all() {
  for (i=0; i < video_class.length; i++) {
    video_class[i].muted = true;
  };
}

(function () {
  for (i=0; i < video_class.length; i++) {
    video_class[i].addEventListener('click', event => {
      if(event.target.muted) {
        mute_all()
        event.target.muted = false;
        event.target.currentTime = 0;
      } else {
        event.target.muted = true;
      }
    })
  }
})()
